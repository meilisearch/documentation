```bash cURL
curl \
  -X POST 'MEILISEARCH_URL/indexes/books/facet-search' \
  -H 'Authorization: Bearer API_KEY' \
  -H 'Content-Type: application/json' \
  --data-binary '{
    "facetName": "genres",
    "facetQuery": "c",
    "exhaustiveFacetCount": true
  }'
```

```bash CLI
meili facet-search \
  --project-url MEILISEARCH_URL --project-api-key API_KEY \
  --index books \
  --facet-name genres --facet-query c --exhaustive-facet-count
```
