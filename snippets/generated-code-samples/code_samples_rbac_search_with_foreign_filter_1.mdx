```bash cURL
curl \
  -X POST 'MEILISEARCH_URL/indexes/documents/search' \
  -H 'Authorization: Bearer TENANT_TOKEN' \
  -H 'Content-Type: application/json' \
  --data-binary '{
    "q": "roadmap",
    "filter": "_foreign(access, user = \"jeremy@meilisearch.com\" OR teams IN [\"product\", \"engineering\"])"
  }'
```

```bash CLI
meili config set --project PROJECT_NAME \
  --project-url MEILISEARCH_URL --project-api-key TENANT_TOKEN

meili search --project PROJECT_NAME --index documents \
  --q roadmap \
  --filter '_foreign(access, user = "jeremy@meilisearch.com" OR teams IN ["product", "engineering"])'
```
