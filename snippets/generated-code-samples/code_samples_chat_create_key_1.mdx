```bash cURL
curl \
  -X POST 'MEILISEARCH_URL/keys' \
  -H 'Authorization: Bearer MEILISEARCH_KEY' \
  -H 'Content-Type: application/json' \
  --data-binary '{
    "name": "Chat API Key",
    "description": "API key for chat completions",
    "actions": ["search", "chatCompletions"],
    "indexes": ["*"],
    "expiresAt": null
  }'
```

```bash CLI
meili config set --project PROJECT_NAME \
  --project-url MEILISEARCH_URL --project-api-key MEILISEARCH_KEY

meili key add --project PROJECT_NAME \
  --name 'Chat API Key' --description 'API key for chat completions' \
  --actions search,chatCompletions --indexes '*' --expires-at null
```
