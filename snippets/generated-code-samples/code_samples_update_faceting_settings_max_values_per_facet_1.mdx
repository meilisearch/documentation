```bash cURL
curl \
  -X PATCH 'MEILISEARCH_URL/indexes/books/settings/faceting' \
  -H 'Authorization: Bearer API_KEY' \
  -H 'Content-Type: application/json' \
  --data-binary '{
```

```bash CLI
meili config set --project PROJECT_NAME \
  --project-url MEILISEARCH_URL --project-api-key API_KEY

meili settings update --project PROJECT_NAME --index books \
  --faceting '{"maxValuesPerFacet":20}'
```
