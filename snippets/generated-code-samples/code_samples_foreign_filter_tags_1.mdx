```bash cURL
curl \
  -X POST 'MEILISEARCH_URL/indexes/articles/search' \
  -H 'Authorization: Bearer API_KEY' \
  -H 'Content-Type: application/json' \
  --data-binary '{
    "q": "cryptography",
    "filter": "_foreign(tags, name = \"security\" AND name = \"encryption\")"
  }'
```

```bash CLI
meili config set --project PROJECT_NAME \
  --project-url MEILISEARCH_URL --project-api-key API_KEY

meili search --project PROJECT_NAME --index articles \
  --q cryptography \
  --filter '_foreign(tags, name = "security" AND name = "encryption")'
```
