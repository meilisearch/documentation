  -H 'Authorization: Bearer API_KEY' \
  -H 'Content-Type: application/json' \
  --data-binary '{
    "embedders": {
      "EMBEDDER_NAME": {

meili config set --project PROJECT_NAME \
  --project-url MEILISEARCH_URL --project-api-key API_KEY

meili settings update --project PROJECT_NAME --index INDEX_NAME \
  --embedders '{
    "EMBEDDER_NAME": {
      "source": "userProvided",
      "dimensions": MODEL_DIMENSIONS
    }
  }'
