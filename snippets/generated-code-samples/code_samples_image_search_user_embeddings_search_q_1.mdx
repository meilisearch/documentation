```bash cURL
curl \
  -X POST 'MEILISEARCH_URL/indexes/INDEX_NAME/search' \
  -H 'Authorization: Bearer API_KEY' \
  -H 'Content-Type: application/json' \
  --data-binary '{
    "vector": VECTORIZED_QUERY,
    "hybrid": { "embedder": "EMBEDDER_NAME" },
    "q": "QUERY"
  }'
```

```bash CLI
meili config set --project PROJECT_NAME \
  --project-url MEILISEARCH_URL --project-api-key API_KEY

meili search --project PROJECT_NAME --index INDEX_NAME \
  --vector VECTORIZED_QUERY --hybrid '{"embedder":"EMBEDDER_NAME"}' --q QUERY
```
