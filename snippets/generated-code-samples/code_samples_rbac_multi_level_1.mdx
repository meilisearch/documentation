```bash cURL
curl \
  -X POST 'MEILISEARCH_URL/indexes/documents/search' \
  -H 'Authorization: Bearer API_KEY' \
  -H 'Content-Type: application/json' \
  --data-binary '{
    "q": "sensitive",
    "filter": "_foreign(access, (user = \"jeremy@meilisearch.com\" AND roles IN [\"editor\", \"owner\"]) OR (teams IN [\"product\", \"engineering\"] AND roles IN [\"editor\"]))"
  }'
```

```bash CLI
meili config set --project PROJECT_NAME \
  --project-url MEILISEARCH_URL --project-api-key API_KEY

meili search --project PROJECT_NAME --index documents \
  --q sensitive \
  --filter '_foreign(access, (user = "jeremy@meilisearch.com" AND roles IN ["editor", "owner"]) OR (teams IN ["product", "engineering"] AND roles IN ["editor"]))'
```
