```bash cURL
  -H 'Authorization: Bearer MEILISEARCH_KEY' \
  --data-binary '{
    "embedders": {
      "default": {
        "source": "openAi",
        "model": "text-embedding-3-small",
```

```bash CLI
meili config set --project PROJECT_NAME \
  --project-url MEILISEARCH_URL --project-api-key MEILISEARCH_KEY

meili settings update --project PROJECT_NAME --index INDEX_NAME \
  --embedders '{
    "default": {
      "source": "openAi",
      "model": "text-embedding-3-small",
      "distribution": { "mean": 0.7, "sigma": 0.3 }
```
