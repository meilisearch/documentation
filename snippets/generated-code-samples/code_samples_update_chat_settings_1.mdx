```bash cURL
  -H 'Authorization: Bearer API_KEY' \
  -H 'Content-Type: application/json' \
  --data-binary '{
```

```bash CLI
meili config set --project PROJECT_NAME \
  --project-url MEILISEARCH_URL --project-api-key API_KEY

meili settings update --project PROJECT_NAME --index INDEX_NAME \
  --chat '{
    "description": "An index of movies.",
    "documentTemplate": "A movie titled '\''{{doc.title}}'\'' released in {{ doc.release_date }}",
    "documentTemplateMaxBytes": 400
  }'
```
