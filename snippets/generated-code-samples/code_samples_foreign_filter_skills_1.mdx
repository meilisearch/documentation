```bash cURL
curl \
  -X POST 'MEILISEARCH_URL/indexes/employees/search' \
  -H 'Authorization: Bearer API_KEY' \
  -H 'Content-Type: application/json' \
  --data-binary '{
    "q": "fullstack",
    "filter": "_foreign(skills, (language = \"Python\" AND proficiency = \"advanced\") OR (language = \"JavaScript\" AND proficiency = \"advanced\"))"
  }'
```

```bash CLI
meili config set --project PROJECT_NAME \
  --project-url MEILISEARCH_URL --project-api-key API_KEY

meili search --project PROJECT_NAME --index employees \
  --q fullstack \
  --filter '_foreign(skills, (language = "Python" AND proficiency = "advanced") OR (language = "JavaScript" AND proficiency = "advanced"))'
```
