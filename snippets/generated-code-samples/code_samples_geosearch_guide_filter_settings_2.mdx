```bash cURL
curl \
  -X PUT 'MEILISEARCH_URL/indexes/neighborhoods/settings/filterable-attributes' \
  -H 'Authorization: Bearer API_KEY' \
  -H 'Content-type:application/json' \
  --data-binary '["_geojson"]'
```

```bash CLI
meili config set --project PROJECT_NAME \
  --project-url MEILISEARCH_URL --project-api-key API_KEY

meili settings update --project PROJECT_NAME --index neighborhoods \
  --filterable-attributes _geojson
```
